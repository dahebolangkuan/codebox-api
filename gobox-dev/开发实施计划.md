# GoBox 开发实施计划

## 项目概述

基于对 `codebox-api` Python 项目的深入分析，本计划详细描述使用 Golang 重构和开发新系统 `gobox` 的完整实施方案。

### 项目定位

GoBox 是一个高性能的沙盒代码执行编排器，为 LLM 应用提供安全、隔离的代码执行环境。相比原 Python 实现，Go 版本将专注于：

- **高并发**: 利用 Goroutine 支持数千并发会话
- **容器池化**: 预热容器池实现毫秒级响应
- **云原生**: 原生支持 Docker/Kubernetes 编排
- **单二进制部署**: 无依赖、跨平台部署

#### 2.1.1 Sandbox 接口

```go
package gobox

import (
    "context"
    "io"
)

// Sandbox 定义沙盒执行环境的标准接口
type Sandbox interface {
    // Exec 执行代码并返回完整结果
    Exec(ctx context.Context, code string, opts ...ExecOption) (*ExecResult, error)
    
    // StreamExec 流式执行代码，实时返回输出
    StreamExec(ctx context.Context, code string, opts ...ExecOption) (<-chan ExecChunk, <-chan error)
    
    // Upload 上传文件到沙盒环境
    Upload(ctx context.Context, files ...File) error
    
    // Download 从沙盒下载文件
    Download(ctx context.Context, filename string) (io.ReadCloser, error)
    
    // ListFiles 列出沙盒中的文件
    ListFiles(ctx context.Context) ([]FileInfo, error)
    
    // Install 安装 Python 包
    Install(ctx context.Context, packages ...string) error
    
    // HealthCheck 健康检查
    HealthCheck(ctx context.Context) error
    
    // Restart 重启执行环境
    Restart(ctx context.Context) error
    
    // Close 关闭并清理资源
    Close() error
}

// ExecOption 执行选项
type ExecOption func(*ExecConfig)

type ExecConfig struct {
    Kernel  string        // "python" 或 "bash"
    Timeout time.Duration // 执行超时
    Cwd     string        // 工作目录
}

func WithKernel(kernel string) ExecOption {
    return func(c *ExecConfig) { c.Kernel = kernel }
}

func WithTimeout(d time.Duration) ExecOption {
    return func(c *ExecConfig) { c.Timeout = d }
}

func WithCwd(cwd string) ExecOption {
    return func(c *ExecConfig) { c.Cwd = cwd }
}
```

#### 2.1.2 类型定义

```go
package gobox

import "time"

// ExecChunk 流式输出块
type ExecChunk struct {
    Type    ChunkType `json:"type"`
    Content string    `json:"content"`
}

type ChunkType string

const (
    ChunkTypeText  ChunkType = "txt"
    ChunkTypeImage ChunkType = "img"
    ChunkTypeError ChunkType = "err"
)

// ExecResult 执行结果
type ExecResult struct {
    Chunks    []ExecChunk   `json:"chunks"`
    Duration  time.Duration `json:"duration"`
    ExitCode  int           `json:"exit_code,omitempty"`
}

// Text 获取所有文本输出
func (r *ExecResult) Text() string {
    var text string
    for _, chunk := range r.Chunks {
        if chunk.Type == ChunkTypeText {
            text += chunk.Content
        }
    }
    return text
}

// Images 获取所有图片 (base64)
func (r *ExecResult) Images() []string {
    var images []string
    for _, chunk := range r.Chunks {
        if chunk.Type == ChunkTypeImage {
            images = append(images, chunk.Content)
        }
    }
    return images
}

// Errors 获取所有错误信息
func (r *ExecResult) Errors() []string {
    var errors []string
    for _, chunk := range r.Chunks {
        if chunk.Type == ChunkTypeError {
            errors = append(errors, chunk.Content)
        }
    }
    return errors
}

// File 上传文件
type File struct {
    Name    string
    Content io.Reader
}

// FileInfo 文件信息
type FileInfo struct {
    Path string `json:"path"`
    Size int64  `json:"size"`
}
```

```go
func (d *Driver) Exec(ctx context.Context, code string, opts ...gobox.ExecOption) (*gobox.ExecResult, error) {
    chunks := make([]gobox.ExecChunk, 0)
    chunkChan, errChan := d.StreamExec(ctx, code, opts...)
    
    for {
        select {
        case chunk, ok := <-chunkChan:
            if !ok {
                return &gobox.ExecResult{Chunks: chunks}, nil
            }
            chunks = append(chunks, chunk)
        case err := <-errChan:
            if err != nil {
                return nil, err
            }
        case <-ctx.Done():
            return nil, ctx.Err()
        }
    }
}
```

#### 2.2.2 容器池实现

```go
package docker

import (
    "context"
    "sync"
    "time"
    
    "github.com/docker/docker/client"
)

// Pool 容器预热池
type Pool struct {
    cli         *client.Client
    config      *PoolConfig
    available   chan *Container  // 可用容器通道
    inUse       map[string]*Container
    mu          sync.RWMutex
    ctx         context.Context
    cancel      context.CancelFunc
}

type PoolConfig struct {
    MinSize     int           // 最小池大小
    MaxSize     int           // 最大池大小
    IdleTimeout time.Duration // 空闲超时
    Image       string        // 容器镜像
    WarmupCount int           // 预热数量
}

func NewPool(config *PoolConfig) (*Pool, error) {
    cli, err := client.NewClientWithOpts(client.FromEnv)
    if err != nil {
        return nil, err
    }
    
    ctx, cancel := context.WithCancel(context.Background())
    
    p := &Pool{
        cli:       cli,
        config:    config,
        available: make(chan *Container, config.MaxSize),
        inUse:     make(map[string]*Container),
        ctx:       ctx,
        cancel:    cancel,
    }
    
    // 启动池维护协程
    go p.maintain()
    
    // 预热容器
    go p.warmup()
    
    return p, nil
}

// Get 获取一个就绪的容器
func (p *Pool) Get(ctx context.Context) (*Container, error) {
    select {
    case container := <-p.available:
        p.mu.Lock()
        p.inUse[container.ID] = container
        p.mu.Unlock()
        return container, nil
    case <-ctx.Done():
        return nil, ctx.Err()
    case <-time.After(30 * time.Second):
        return nil, gobox.ErrPoolExhausted
    }
}

// Release 释放容器
func (p *Pool) Release(container *Container, reuse bool) {
    p.mu.Lock()
    delete(p.inUse, container.ID)
    p.mu.Unlock()
    
    if reuse && container.IsHealthy() {
        // 重置容器状态后放回池中
        container.Reset()
        select {
        case p.available <- container:
        default:
            // 池已满，销毁容器
            container.Destroy()
        }
    } else {
        container.Destroy()
    }
}

// maintain 维护池大小
func (p *Pool) maintain() {
    ticker := time.NewTicker(10 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-p.ctx.Done():
            return
        case <-ticker.C:
            p.mu.RLock()
            currentSize := len(p.available) + len(p.inUse)
            p.mu.RUnlock()
            
            // 补充容器
            for currentSize < p.config.MinSize {
                if container, err := p.createContainer(); err == nil {
                    select {
                    case p.available <- container:
                        currentSize++
                    default:
                    }
                }
            }
        }
    }
}

// warmup 预热容器
func (p *Pool) warmup() {
    for i := 0; i < p.config.WarmupCount; i++ {
        if container, err := p.createContainer(); err == nil {
            p.available <- container
        }
    }
}

// Close 关闭池
func (p *Pool) Close() error {
    p.cancel()
    
    // 清理所有容器
    close(p.available)
    for container := range p.available {
        container.Destroy()
    }
    
    p.mu.Lock()
    for _, container := range p.inUse {
        container.Destroy()
    }
    p.mu.Unlock()
    
    return nil
}
```

```go
package remote

import (
    "bufio"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"
    
    "gobox/pkg/gobox"

    baseURL    string
    apiKey     string
    factoryID  string
    sessionID  string
    httpClient *http.Client
}

type Config struct {
    BaseURL   string
    APIKey    string
    FactoryID string
    SessionID string
    Timeout   time.Duration
}

func New(config *Config) *Driver {
    return &Driver{
        baseURL:   config.BaseURL,
        apiKey:    config.APIKey,
        factoryID: config.FactoryID,
        sessionID: config.SessionID,
        httpClient: &http.Client{
            Timeout: config.Timeout,
        },
    }
}

func (d *Driver) StreamExec(ctx context.Context, code string, opts ...gobox.ExecOption) (<-chan gobox.ExecChunk, <-chan error) {
    chunkChan := make(chan gobox.ExecChunk)
    errChan := make(chan error, 1)
    
    go func() {
        defer close(chunkChan)
        defer close(errChan)
        
        // 构建请求
        url := fmt.Sprintf("%s/codebox/%s/exec", d.baseURL, d.sessionID)
        req, err := http.NewRequestWithContext(ctx, "POST", url, nil)
        if err != nil {
            errChan <- err
            return
        }
        
        req.Header.Set("Authorization", "Bearer "+d.apiKey)
        req.Header.Set("Factory-Id", d.factoryID)
        
        resp, err := d.httpClient.Do(req)
        if err != nil {
            errChan <- err
            return
        }
        defer resp.Body.Close()
        
        // 解析流式响应 (JSONL)
        scanner := bufio.NewScanner(resp.Body)
        for scanner.Scan() {
            var chunk gobox.ExecChunk
            if err := json.Unmarshal(scanner.Bytes(), &chunk); err != nil {
                // 尝试解析旧协议格式
                chunk = parseProtocol(scanner.Text())
            }
            
            select {
            case chunkChan <- chunk:
            case <-ctx.Done():
                return
            }
        }
    }()
    
    return chunkChan, errChan
}
```

### 2.4 协议处理 (pkg/protocol)

```go
package protocol

import (
    "encoding/json"
    "regexp"
    
    "gobox/pkg/gobox"
)

// 旧协议正则 <txt>content</txt>
var legacyPattern = regexp.MustCompile(`<(txt|img|err)>(.*?)</\1>`)

// ParseLegacy 解析旧协议格式
func ParseLegacy(data string) []gobox.ExecChunk {
    matches := legacyPattern.FindAllStringSubmatch(data, -1)
    chunks := make([]gobox.ExecChunk, 0, len(matches))
    
    for _, match := range matches {
        chunks = append(chunks, gobox.ExecChunk{
            Type:    gobox.ChunkType(match[1]),
            Content: match[2],
        })
    }
    
    return chunks
}

// EncodeJSONL 编码为 JSONL 格式
func EncodeJSONL(chunk gobox.ExecChunk) ([]byte, error) {
    return json.Marshal(chunk)
}

// DecodeJSONL 解码 JSONL 格式
func DecodeJSONL(data []byte) (gobox.ExecChunk, error) {
    var chunk gobox.ExecChunk
    err := json.Unmarshal(data, &chunk)
    return chunk, err
}
```

```go
    s.mux.HandleFunc("POST /files/upload", s.handleUpload)
}

func (s *Server) handleHealth(w http.ResponseWriter, r *http.Request) {
    json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
}

func (s *Server) handleExec(w http.ResponseWriter, r *http.Request) {
    var req struct {
        Code    string `json:"code"`
        Kernel  string `json:"kernel"`
        Timeout int    `json:"timeout,omitempty"`
        Cwd     string `json:"cwd,omitempty"`
    }
    
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, err.Error(), http.StatusBadRequest)
        return
    }
    
    // 设置流式响应头
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    
    flusher, ok := w.(http.Flusher)
    if !ok {
        http.Error(w, "Streaming not supported", http.StatusInternalServerError)
        return
    }
    
    opts := []gobox.ExecOption{gobox.WithKernel(req.Kernel)}
    if req.Timeout > 0 {
        opts = append(opts, gobox.WithTimeout(time.Duration(req.Timeout)*time.Second))
    }
    
    chunkChan, errChan := s.sandbox.StreamExec(r.Context(), req.Code, opts...)
    
    for {
        select {
        case chunk, ok := <-chunkChan:
            if !ok {
                return
            }
            // 发送 JSONL 格式
            data, _ := json.Marshal(chunk)
            w.Write(data)
            w.Write([]byte("\n"))
            flusher.Flush()
        case err := <-errChan:
            if err != nil {
                http.Error(w, err.Error(), http.StatusInternalServerError)
            }
            return
        case <-r.Context().Done():
            return
        }
    }
}

func (s *Server) ListenAndServe() error {
    server := &http.Server{
        Addr:         fmt.Sprintf(":%d", s.config.Port),
        Handler:      s.mux,
        ReadTimeout:  s.config.ReadTimeout,
        WriteTimeout: s.config.WriteTimeout,
        IdleTimeout:  s.config.IdleTimeout,
    }
    
    return server.ListenAndServe()
}
```

## 4. 技术规范

### 4.1 依赖管理

```go
// go.mod
module gobox

go 1.22

require (
    github.com/docker/docker v24.0.7+incompatible
    github.com/spf13/cobra v1.8.0
    github.com/stretchr/testify v1.8.4
    go.uber.org/zap v1.26.0
)
```

### 4.2 代码规范

- 遵循 [Effective Go](https://golang.org/doc/effective_go)
- 使用 `gofmt` 格式化代码
- 使用 `golangci-lint` 进行静态检查
- 所有导出的函数/类型必须有文档注释
- 错误处理遵循 Go 惯例，不使用 panic

### 4.3 日志规范

```go
import "go.uber.org/zap"

var logger *zap.Logger

func init() {
    logger, _ = zap.NewProduction()
}

// 使用结构化日志
logger.Info("container started",
    zap.String("container_id", id),
    zap.Duration("startup_time", duration),
)
```

### 4.4 并发规范

- 使用 `context.Context` 控制超时和取消
- 使用 `sync.Mutex` 或 `sync.RWMutex` 保护共享状态
- Channel 使用带缓冲的通道，避免 goroutine 泄漏
- 使用 `errgroup` 管理并发任务

### 4.5 API 规范

- RESTful API 设计
- 错误响应格式统一:
```json
{
    "error": {
        "code": "EXECUTION_TIMEOUT",
        "message": "Code execution exceeded timeout",
        "details": {}
    }
}
```
